<script>
const pairs = [
["ا","1-با"],["ب","2-تا"],["ت","3-ثا"],["ث","4-جا"],
["ج","5-حا"],["ح","6-خا"],["خ","7-دا"],["د","8-ذا"],
["ذ","9-را"],["ر","0-زا"],["ز","1-سا"],["س","2-شا"],
["ش","3-صا"],["ص","4-ضا"],["ض","5-طا"],["ط","6-ظا"],
["ظ","7-عا"],["ع","8-غا"],["غ","9-فا"],["ف","0-قا"],
["ق","1-كا"],["ك","2-لا"],["ل","3-ما"],["م","4-نا"],
["ن","5-ها"],["ه","6-وا"],["و","7-يا"],["ي","8-أا"],
["ء","9-أب"],["ة","0-أت"],["ى","1-أث"],["أ","2-أج"],
["إ","3-أح"],["آ","4-أخ"]
];
const encodeMap = new Map(pairs);
const decodeMap = new Map(pairs.map(([letter,code])=>[code,letter]));

function encrypt(){
 const text=document.getElementById("plain").value.normalize("NFC");
 const out=[];
 for(const ch of text){
   if(ch===" "){out.push("/");continue;}
   if(encodeMap.has(ch)){out.push(encodeMap.get(ch));continue;}
   if(ch==="\n"){out.push("[سطر]");continue;}
   out.push("[حرف:"+ch.codePointAt(0).toString(16)+"]");
 }
 document.getElementById("encoded").textContent=out.join(" ");
}
function decrypt(){
 const tokens=document.getElementById("cipher").value.trim().split(/\s+/);
 const out=[];
 for(const token of tokens){
   if(!token)continue;
   if(token==="/"){out.push(" ");continue;}
   if(token==="[سطر]"){out.push("\n");continue;}
   if(decodeMap.has(token)){out.push(decodeMap.get(token));continue;}
   const m=token.match(/^\[حرف:([0-9a-f]+)\]$/i);
   if(m){
     const cp=parseInt(m[1],16);
     if(Number.isFinite(cp)&&cp<=0x10ffff)out.push(String.fromCodePoint(cp));
     else out.push("�");
   }else out.push("�");
 }
 document.getElementById("decoded").textContent=out.join("");
}
function clearAll(){
 ["plain","encoded","cipher","decoded"].forEach(id=>document.getElementById(id).value!==undefined?document.getElementById(id).value="":document.getElementById(id).textContent="");
}
async function copyText(id){
 const text=document.getElementById(id).textContent;
 try{await navigator.clipboard.writeText(text);alert("تم النسخ!");}
 catch(e){alert("تعذّر النسخ التلقائي. حدّد النص وانسخه يدويًا.");}
}
function renderTable(){
 const q=document.getElementById("search").value.trim();
 const body=document.getElementById("tableBody");
 body.replaceChildren();
 pairs.filter(([letter,code])=>!q||letter.includes(q)||code.includes(q))
 .forEach(([letter,code])=>{
   const tr=document.createElement("tr");
   const td1=document.createElement("td"),td2=document.createElement("td");
   td1.textContent=letter;td2.textContent=code;
   tr.append(td1,td2);body.append(tr);
 });
}
renderTable();
